# Root = country. Place bet BET (1-5), spin, pay out and show the result.
# Result events (one per colour, so the text can name it):
#   .2 win on red, .5 win on black, .4 zero hit,
#   .3 lose on red, .6 lose on black, .7 lose on green (zero)
fallout_vav_rl_play_effect = {
	set_variable = { name = vav_rl_bet value = $BET$ }
	fallout_vav_rl_spin_effect = yes
	if = {
		limit = {
			OR = {
				AND = { var:vav_rl_bet = 1 var:vav_rl_color = 1 }
				AND = { var:vav_rl_bet = 2 var:vav_rl_color = 2 }
				AND = { var:vav_rl_bet = 3 var:vav_rl_parity = 1 }
				AND = { var:vav_rl_bet = 4 var:vav_rl_parity = 2 }
			}
		}
		fallout_vav_bj_fortune_effect = { AMOUNT = 10 }
		if = {
			limit = { var:vav_rl_color = 1 }
			trigger_event_non_silently = vav_roulette.2	# win, red
		}
		else = {
			trigger_event_non_silently = vav_roulette.5	# win, black
		}
	}
	else_if = {
		limit = { var:vav_rl_bet = 5 var:vav_rl_number = 0 }
		fallout_vav_bj_fortune_effect = { AMOUNT = 35 }
		trigger_event_non_silently = vav_roulette.4	# zero hit
	}
	else = {
		fallout_vav_bj_fortune_effect = { AMOUNT = -10 }
		if = {
			limit = { var:vav_rl_color = 1 }
			trigger_event_non_silently = vav_roulette.3	# lose, red
		}
		else_if = {
			limit = { var:vav_rl_color = 2 }
			trigger_event_non_silently = vav_roulette.6	# lose, black
		}
		else = {
			trigger_event_non_silently = vav_roulette.7	# lose, green
		}
	}
}
