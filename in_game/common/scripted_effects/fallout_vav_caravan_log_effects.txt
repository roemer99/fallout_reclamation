# =============================================================================
# FALLOUT RECLAMATION - Vault 21: Caravan move log (effects)
# =============================================================================
# Shown by the "Log" button of the Caravan window (gui/fallout_caravan_log.gui).
# Per turn t (1-9) and player W (p = player, o = opponent):
#   vav_cv_lg_W_c_t  card (1-14, 0 = no entry)
#   vav_cv_lg_W_s_t  side: 0 = discarded, 1 = player's caravan, 2 = opponent's caravan
#   vav_cv_lg_W_l_t  caravan (1-3)
# A move is first written to vav_cv_lg_card / _side / _lane, then stored under
# the current turn (vav_cv_turn).
# =============================================================================

# Root = country. Clear the whole log (new game).
fallout_vav_cv_log_reset_effect = {
	fallout_vav_cv_log_reset_turn_effect = { T = 1 }
	fallout_vav_cv_log_reset_turn_effect = { T = 2 }
	fallout_vav_cv_log_reset_turn_effect = { T = 3 }
	fallout_vav_cv_log_reset_turn_effect = { T = 4 }
	fallout_vav_cv_log_reset_turn_effect = { T = 5 }
	fallout_vav_cv_log_reset_turn_effect = { T = 6 }
	fallout_vav_cv_log_reset_turn_effect = { T = 7 }
	fallout_vav_cv_log_reset_turn_effect = { T = 8 }
	fallout_vav_cv_log_reset_turn_effect = { T = 9 }
}
fallout_vav_cv_log_reset_turn_effect = {
	set_variable = { name = vav_cv_lg_p_c_$T$ value = 0 }
	set_variable = { name = vav_cv_lg_p_s_$T$ value = 0 }
	set_variable = { name = vav_cv_lg_p_l_$T$ value = 0 }
	set_variable = { name = vav_cv_lg_o_c_$T$ value = 0 }
	set_variable = { name = vav_cv_lg_o_s_$T$ value = 0 }
	set_variable = { name = vav_cv_lg_o_l_$T$ value = 0 }
}

# Root = country. Store vav_cv_lg_card / _side / _lane for player WHO (p / o) in the current turn.
fallout_vav_cv_log_store_effect = {
	if = { limit = { var:vav_cv_turn = 1 } fallout_vav_cv_log_store_turn_effect = { WHO = $WHO$ T = 1 } }
	if = { limit = { var:vav_cv_turn = 2 } fallout_vav_cv_log_store_turn_effect = { WHO = $WHO$ T = 2 } }
	if = { limit = { var:vav_cv_turn = 3 } fallout_vav_cv_log_store_turn_effect = { WHO = $WHO$ T = 3 } }
	if = { limit = { var:vav_cv_turn = 4 } fallout_vav_cv_log_store_turn_effect = { WHO = $WHO$ T = 4 } }
	if = { limit = { var:vav_cv_turn = 5 } fallout_vav_cv_log_store_turn_effect = { WHO = $WHO$ T = 5 } }
	if = { limit = { var:vav_cv_turn = 6 } fallout_vav_cv_log_store_turn_effect = { WHO = $WHO$ T = 6 } }
	if = { limit = { var:vav_cv_turn = 7 } fallout_vav_cv_log_store_turn_effect = { WHO = $WHO$ T = 7 } }
	if = { limit = { var:vav_cv_turn = 8 } fallout_vav_cv_log_store_turn_effect = { WHO = $WHO$ T = 8 } }
	if = { limit = { var:vav_cv_turn = 9 } fallout_vav_cv_log_store_turn_effect = { WHO = $WHO$ T = 9 } }
}
fallout_vav_cv_log_store_turn_effect = {
	set_variable = { name = vav_cv_lg_$WHO$_c_$T$ value = var:vav_cv_lg_card }
	set_variable = { name = vav_cv_lg_$WHO$_s_$T$ value = var:vav_cv_lg_side }
	set_variable = { name = vav_cv_lg_$WHO$_l_$T$ value = var:vav_cv_lg_lane }
}

# Root = country. The player's move: card vav_cv_card, SIDE / LANE as numbers.
fallout_vav_cv_log_player_effect = {
	set_variable = { name = vav_cv_lg_card value = var:vav_cv_card }
	set_variable = { name = vav_cv_lg_side value = $SIDE$ }
	set_variable = { name = vav_cv_lg_lane value = $LANE$ }
	fallout_vav_cv_log_store_effect = { WHO = p }
}

# Root = country. The opponent's move (called at the end of its move):
#   number card placed -> its caravan vav_cv_olane, face card -> target vav_cv_f2t,
#   otherwise discarded.
fallout_vav_cv_log_opponent_effect = {
	set_variable = { name = vav_cv_lg_card value = var:vav_cv_ocard }
	set_variable = { name = vav_cv_lg_side value = 0 }
	set_variable = { name = vav_cv_lg_lane value = 0 }
	if = {
		limit = { var:vav_cv_ocard <= 10 var:vav_cv_done = 1 }
		set_variable = { name = vav_cv_lg_side value = 2 }
		set_variable = { name = vav_cv_lg_lane value = var:vav_cv_olane }
	}
	else_if = {
		limit = { var:vav_cv_ocard >= 11 var:vav_cv_f2t >= 1 var:vav_cv_f2t <= 3 }
		set_variable = { name = vav_cv_lg_side value = 1 }
		set_variable = { name = vav_cv_lg_lane value = var:vav_cv_f2t }
	}
	else_if = {
		limit = { var:vav_cv_ocard >= 11 var:vav_cv_f2t >= 4 }
		set_variable = { name = vav_cv_lg_side value = 2 }
		set_variable = { name = vav_cv_lg_lane value = var:vav_cv_f2t }
		change_variable = { name = vav_cv_lg_lane add = -3 }
	}
	fallout_vav_cv_log_store_effect = { WHO = o }
}
