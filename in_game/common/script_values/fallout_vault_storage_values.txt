# =============================================================================
# FALLOUT RECLAMATION - Vault storage: display values (REAL market stockpile)
# =============================================================================
# Country scope = player. Reads the stockpile of the Vault market
# (capital.market, link with data: "capital.market.stockpile_in_market(goods:<goods>)",
# same function-style syntax as vanilla "goods_supply_in_market(goods:silver)").
# Years left = stock / yearly need of the Vault buildings.
# =============================================================================

fallout_vault_stock_glass_display = {
	value = 0
	if = {
		limit = { exists = capital.market }
		value = "capital.market.stockpile_in_market(goods:glass)"
	}
}
fallout_vault_stock_glass_years = {
	value = fallout_vault_stock_glass_display
	divide = 0.2	# yearly need of the Vault buildings
}
