# =============================================================================
# FALLOUT RECLAMATION - Vault 21: Blackjack table (scripted GUI)
# =============================================================================
# Button in gui/panels/situation/vault_management.gui (Vault 21 window):
#   onclick = "[GetScriptedGui('vav_blackjack_open').Execute(GuiScope.SetRoot(GetPlayer.MakeScope).End)]"
# Deals a new hand and opens the Blackjack event (events/vaults/vav_blackjack.txt).
# =============================================================================

vav_blackjack_open = {
	scope = country

	is_shown = {
		tag = VAV
	}

	is_valid = {
		tag = VAV
		custom_tooltip = {
			text = vav_blackjack_cooldown_tt
			NOT = { has_variable = vav_bj_cooldown }
		}
	}

	effect = {
		root = {
			fallout_vav_bj_new_game_effect = yes
			# Natural blackjack (21 with the first two cards) wins at once
			if = {
				limit = { var:vav_bj_player = 21 }
				trigger_event_non_silently = vav_blackjack.5
			}
			else = {
				trigger_event_non_silently = vav_blackjack.1
			}
		}
	}
}

# Roulette: place a bet in event vav_roulette.1 (events/vaults/vav_roulette.txt)
vav_roulette_open = {
	scope = country

	is_shown = {
		tag = VAV
	}

	is_valid = {
		tag = VAV
		custom_tooltip = {
			text = vav_roulette_cooldown_tt
			NOT = { has_variable = vav_rl_cooldown }
		}
	}

	effect = {
		root = {
			set_variable = { name = vav_rl_cooldown value = 1 days = 30 }
			trigger_event_non_silently = vav_roulette.1
		}
	}
}

# Caravan table: show / hide the move log (gui/fallout_caravan_log.gui)
vav_cv_toggle_log = {
	scope = country
	is_shown = { tag = VAV }
	effect = {
		root = {
			if = {
				limit = { has_variable = vav_cv_show_log }
				remove_variable = vav_cv_show_log
			}
			else = {
				set_variable = vav_cv_show_log
			}
		}
	}
}
